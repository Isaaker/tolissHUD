cmake_minimum_required(VERSION 3.20)
project(XPHUD_Native3D VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT DEFINED XPLANE_SDK_ROOT)
  message(FATAL_ERROR "Set XPLANE_SDK_ROOT to the extracted X-Plane SDK directory.")
endif()
set(XPLM_INCLUDE_DIR "${XPLANE_SDK_ROOT}/CHeaders/XPLM")
if(NOT EXISTS "${XPLM_INCLUDE_DIR}/XPLMPlugin.h")
  message(FATAL_ERROR "XPLMPlugin.h not found in ${XPLM_INCLUDE_DIR}")
endif()
add_library(XPHUD_Native3D SHARED src/plugin.cpp)
target_include_directories(XPHUD_Native3D PRIVATE "${XPLM_INCLUDE_DIR}")
target_compile_definitions(XPHUD_Native3D PRIVATE XPLM200=1 XPLM210=1 XPLM300=1)
set_target_properties(XPHUD_Native3D PROPERTIES PREFIX "" SUFFIX ".xpl")
if(WIN32)
  target_compile_definitions(XPHUD_Native3D PRIVATE IBM=1)
  set_target_properties(XPHUD_Native3D PROPERTIES OUTPUT_NAME "win")
  set(XPLM_IMPORT_LIB "${XPLANE_SDK_ROOT}/Libraries/Win/XPLM_64.lib")
  if(NOT EXISTS "${XPLM_IMPORT_LIB}")
    message(FATAL_ERROR "XPLM_64.lib not found: ${XPLM_IMPORT_LIB}")
  endif()
  target_link_libraries(XPHUD_Native3D PRIVATE "${XPLM_IMPORT_LIB}")
elseif(APPLE)
  target_compile_definitions(XPHUD_Native3D PRIVATE APL=1)
  set_target_properties(XPHUD_Native3D PROPERTIES OUTPUT_NAME "mac")
  target_link_options(XPHUD_Native3D PRIVATE "-undefined" "dynamic_lookup")
elseif(UNIX)
  target_compile_definitions(XPHUD_Native3D PRIVATE LIN=1)
  set_target_properties(XPHUD_Native3D PROPERTIES OUTPUT_NAME "lin")
  target_link_options(XPHUD_Native3D PRIVATE "-Wl,--allow-shlib-undefined")
else()
  message(FATAL_ERROR "Unsupported platform")
endif()
if(MSVC)
  target_compile_options(XPHUD_Native3D PRIVATE /W4 /permissive-)
else()
  target_compile_options(XPHUD_Native3D PRIVATE -Wall -Wextra -Wpedantic)
endif()
